find_package(Qt6 REQUIRED COMPONENTS Test)

# chat_test(<name> [LIBRARY <lib>...]): tests/tst_<name>.cpp as a CTest test.
# Every test runs headless, with its own HOME and XDG folders (TestSupport.h).
function(chat_test name)
    cmake_parse_arguments(ARG "" "" "LIBRARY" ${ARGN})
    if(NOT ARG_LIBRARY)
        set(ARG_LIBRARY chat_core)
    endif()
    qt_add_executable(tst_${name} tst_${name}.cpp TestSupport.h)
    target_link_libraries(tst_${name} PRIVATE ${ARG_LIBRARY} Qt6::Test)
    add_test(NAME ${name} COMMAND tst_${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
endfunction()

chat_test(store)
chat_test(crypto)
chat_test(ratchet)
chat_test(protocol)
chat_test(attachment)
chat_test(voice)
chat_test(server LIBRARY chat_server)
chat_test(messenger LIBRARY chat_server)
chat_test(mainwindow LIBRARY chat_app chat_server)
chat_test(singleinstance LIBRARY chat_app)
